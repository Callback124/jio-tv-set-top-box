  (String,String) _logoUrls(String raw) {
    final clean=raw.trim().replaceFirst(RegExp(r'^/+'),'');
    if(clean.isEmpty)return ('','');
    if(clean.startsWith('http'))return (clean,clean);
    final path=clean.startsWith('dare_images/')?clean:'dare_images/images/'+clean;
    return ('https://jiotvimages.cdn.jio.com/'+path,'https://jiotv.catchup.cdn.jio.com/'+path);

  final List<String> _debugErrors = [];

  Future<List<TvChannel>> _fetchJio() async {
    const urls=[
      'https://jiotv.data.cdn.jio.com/apis/v3.1/getMobileChannelList/get/?os=android&devicetype=phone&usertype=tvYR7NSNn7rymo3F',
      'https://jiotv.data.cdn.jio.com/apis/v3.0/getMobileChannelList/get/?os=android&devicetype=phone&usertype=tvYR7NSNn7rymo3F'
    ];
    for(final url in urls){
      try{
        final response=await http.get(Uri.parse(url),headers:headers).timeout(const Duration(seconds:18));
        if(response.statusCode!=200){
          _debugErrors.add('JioTV API HTTP ${response.statusCode}');
          continue;
        }
        if(response.body.trim().isEmpty){
          _debugErrors.add('JioTV API: empty response body');
          continue;
        }
        final decoded=jsonDecode(response.body);
        final List rows=decoded is Map&&decoded['result'] is List?decoded['result'] as List:[];
        final parsed=rows.whereType<Map>().map((e){
          final j=Map<String,dynamic>.from(e);
          final id=_int(j['channel_id']);
          final logos=_logoUrls((j['logoUrl']??'').toString());
          final c=_int(j['channelCategoryId']),l=_int(j['channelLanguageId']);
          return TvChannel(id:id,name:(j['channel_name']??'Channel '+id.toString()).toString(),
            logo:logos.$1,fallbackLogo:logos.$2,categoryId:c,category:_category(c),
            languageId:l,language:_language(l),hd:j['isHD']==true,
            order:_int(j['channel_order'],999999),raw:j);
        }).where((c)=>c.id>0).toList();
        if(parsed.isNotEmpty)return parsed;
        _debugErrors.add('JioTV API: HTTP 200 but result list empty/unrecognized');
      }catch(e){
        _debugErrors.add('JioTV API error: $e');
      }
    }
    return [];
  }

  Future<List<TvChannel>> _fetchExtra() async {
    try{
      final response=await http.get(Uri.parse(extraUrl),headers:const {
        'Accept':'application/json',
        'User-Agent':'Mozilla/5.0 (Linux; Android 15; Pixel 9) AppleWebKit/537.36 Chrome/150.0.0.0 Mobile Safari/537.36'
      }).timeout(const Duration(seconds:18));
      if(response.statusCode!=200||response.body.trim().isEmpty){
        _debugErrors.add('npoint API HTTP ${response.statusCode} (empty body: ${response.body.trim().isEmpty})');
        return [];
      }
      final decoded=jsonDecode(response.body);
      List rows=[];
      if(decoded is List)rows=decoded;
      if(decoded is Map){
        for(final key in ['channels','result','data','list','items','tv']){
          if(decoded[key] is List){rows=decoded[key] as List;break;}
        }
        if(rows.isEmpty)rows=decoded.values.firstWhere((v)=>v is List,orElse:()=>[]) as List;
      }
      final result=<TvChannel>[];
      for(var i=0;i<rows.length;i++){
        if(rows[i] is! Map)continue;
        final j=Map<String,dynamic>.from(rows[i]);
        String pick(List<String> keys){
          for(final key in keys){
            final value=(j[key]??'').toString().trim();
            if(value.isNotEmpty&&value!='null')return value;
          }
          return '';
        }
        final name=pick(['channel_name','channelName','name','title','tvg_name']);
        final stream=pick(['channel_url','stream_url','mpd_url','m3u8_url','url','ready_to_use_link']);
        if(name.isEmpty&&stream.isEmpty)continue;
        final id=_int(pick(['channel_id','channelId','id','chid','channelNo']),1000000+i);
        final logos=_logoUrls(pick(['channel_logo','logoUrl','logo','tvg_logo','image','thumbnail']));
        final rawCategory=pick(['channel_genre','cat','categoryName','category','genre','group_title']);
        final c=_int(pick(['channelCategoryId','category_id','categoryId']));
        final rawLanguage=pick(['languageName','language','lang']);
        final l=_int(pick(['channelLanguageId','language_id','languageId']));
        final normalizedLanguage=rawLanguage.toLowerCase()=='malyalam'?'Malayalam':rawLanguage;
        final inferredCategory=c!=0?c:_inferCategory(rawCategory+' '+name);
        final inferredLanguage=l!=0?l:_inferLanguage(normalizedLanguage);
        result.add(TvChannel(id:id,name:name.isEmpty?'Channel '+id.toString():name,
          logo:logos.$1,fallbackLogo:logos.$2,categoryId:inferredCategory,
          category:rawCategory.isNotEmpty?rawCategory:_category(inferredCategory),
          languageId:inferredLanguage,language:normalizedLanguage.isNotEmpty?normalizedLanguage:_language(inferredLanguage),
          hd:j['isHD']==true||j['isHd']==true||name.toUpperCase().contains('HD'),
          order:_int(pick(['channel_order','order']),10000+i),raw:j));
      }
      if(result.isEmpty)_debugErrors.add('npoint API: 0 channels parsed (raw=${rows.length})');
      return result;
    }catch(e){
      _debugErrors.add('npoint API error: $e');
      return [];
    }
  }

  int _inferCategory(String value){
    final s=value.toLowerCase();
    if(s.contains('movie')||s.contains('cinema'))return 6;
    if(s.contains('sport')||s.contains('cricket'))return 8;
    if(s.contains('news'))return 12;
    if(s.contains('kid')||s.contains('cartoon')||s.contains('nick'))return 7;
    if(s.contains('music')||s.contains('mtv'))return 13;
    if(s.contains('discovery')||s.contains('history')||s.contains('nat geo'))return 10;
    if(s.contains('bhakti')||s.contains('sanskar')||s.contains('aastha'))return 15;
    return value.trim().isEmpty?0:5;
  }
  int _inferLanguage(String value){
    final s=value.toLowerCase();
    for(final entry in languages.entries){if(entry.key!=0&&s.contains(entry.value.toLowerCase()))return entry.key;}
    return 0;
  }

  Future<void> _loadChannels() async {
    setState(() { loading=true; error=null; _debugErrors.clear(); });
    try{
      final results=await Future.wait([_fetchJio(),_fetchExtra()]);
      final merged=<int,TvChannel>{};
      for(final channel in results[0]){merged.putIfAbsent(channel.id,()=>channel);}
      for(final extra in results[1]){
        final old=merged[extra.id];
        if(old==null){merged[extra.id]=extra;}
        else{
          merged[extra.id]=TvChannel(id:old.id,name:old.name.isNotEmpty?old.name:extra.name,
            logo:old.logo.isNotEmpty?old.logo:extra.logo,
            fallbackLogo:old.fallbackLogo.isNotEmpty?old.fallbackLogo:extra.fallbackLogo,
            categoryId:old.categoryId!=0?old.categoryId:extra.categoryId,
            category:old.categoryId!=0?old.category:extra.category,
            languageId:old.languageId!=0?old.languageId:extra.languageId,
            language:old.languageId!=0?old.language:extra.language,
            hd:old.hd||extra.hd,order:old.order,raw:extra.raw);
        }
      }
      final sorted=merged.values.toList()..sort((a,b){
        final order=a.order.compareTo(b.order);
        return order!=0?order:a.name.toLowerCase().compareTo(b.name.toLowerCase());
      });
      if(sorted.isEmpty){
        final details=_debugErrors.isEmpty?'Both APIs returned zero channels.':_debugErrors.join('\\n');
        throw Exception(details);
      }
      if(!mounted)return;
      setState(()=>allChannels=sorted);
    }catch(e){
      if(mounted)setState(()=>error=e.toString().replaceFirst('Exception: ',''));
    }finally{
      if(mounted)setState(()=>loading=false);
    }
  }

      (q.isEmpty||c.name.toLowerCase().contains(q)||c.category.toLowerCase().contains(q)||
       c.language.toLowerCase().contains(q)||c.id.toString()==q)).toList();
  }

  Future<void> _openChannel(TvChannel channel) async {
    final raw=channel.raw??<String,dynamic>{};
    final rawChannelId=(raw['channel_id']??raw['channelId']??'').toString().trim();
    final requestHeaders=<String,String>{
      'User-Agent':(raw['user_agent']??raw['userAgent']??'Mozilla/5.0 (Linux; Android 15; Pixel 9) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/150.0.0.0 Mobile Safari/537.36').toString(),
      'Accept':'*/*','Connection':'keep-alive',
    };
    final referer=(raw['referer']??raw['referrer']??'').toString().trim();
    final origin=(raw['origin']??'').toString().trim();
    final cookie=(raw['cookie']??'').toString().trim();
    if(referer.isNotEmpty)requestHeaders['Referer']=referer;
    if(origin.isNotEmpty)requestHeaders['Origin']=origin;
    if(cookie.isNotEmpty)requestHeaders['Cookie']=cookie;
    String firstValue(List<String> keys) {
      for(final key in keys) {
        final value=(raw[key]??'').toString().trim();
        if(value.isNotEmpty&&value!='null')return value;
      }
      return '';
    }
    bool isHttp(String value)=>value.startsWith('https://')||value.startsWith('http://');
    final direct=firstValue(['url','stream_url','m3u8_url','mpd_url','play_url','playUrl']);
    final channelUrl=firstValue(['channel_url']);
    final hasResolverId=rawChannelId.isNotEmpty&&channelUrl.isNotEmpty;
    String? streamUrl;
    String drmLicenseUrl='';
    if(isHttp(direct)) {
      streamUrl=direct;
    } else if(!hasResolverId&&isHttp(channelUrl)) {
      streamUrl=channelUrl;
    } else {
      if(rawChannelId.isEmpty) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content:Text('Playable URL / Channel ID missing: '+channel.name)));
        return;
      }
      try {
        final response=await http.get(
          Uri.https('eliteapiyash.streamxlive.workers.dev','/',{'id':rawChannelId}),
          headers:requestHeaders,
        ).timeout(const Duration(seconds:25));
        final body=response.body.replaceFirst('\uFEFF','').trim();
        // Some resolver deployments append a plain-text attribution/update footer
        // after valid JSON. Extract exactly the first complete JSON value before
        // decoding; jsonDecode itself correctly rejects trailing non-JSON text.

        if(response.statusCode!=200) {
          final preview=body.substring(0,body.length.clamp(0,220));
          throw Exception('Resolver HTTP '+response.statusCode.toString()+': '+preview);
        }
        if(body.isEmpty)throw Exception('Resolver returned an empty response.');
        dynamic decoded;
        try {
          decoded=jsonDecode(_extractJsonValue(body));
        } on FormatException {
          if(isHttp(body.split('|').first.trim())) {
            decoded={'ready_to_use_link':body,'status':'success'};
          } else {
            final preview=body.substring(0,body.length.clamp(0,260));
            throw Exception('Resolver JSON invalid ('+(response.headers['content-type']??'unknown content-type')+'). Response: '+preview);
          }
        }
        final Map<String,dynamic> data=decoded is List&&decoded.isNotEmpty&&decoded.first is Map
          ?Map<String,dynamic>.from(decoded.first as Map)
          :decoded is Map?Map<String,dynamic>.from(decoded as Map):<String,dynamic>{};
        if((data['status']??'success').toString().toLowerCase()=='error') {
          throw Exception((data['message']??'Stream resolve failed').toString());
        }
        drmLicenseUrl=firstFromMap(data,['widevine_license_url','drm_license_url','license_url'])??'';
        final ready=(data['ready_to_use_link']??'').toString().trim();
        if(ready.isNotEmpty) {
          final parts=ready.split('|');
          final candidate=parts.first.trim();
          if(isHttp(candidate))streamUrl=candidate;
          final options=parts.skip(1).join('|');
          final cookieMatch=RegExp(r'(?:^|&)cookie=([^&]+)').firstMatch(options);
          if(cookieMatch!=null)requestHeaders['Cookie']=Uri.decodeComponent(cookieMatch.group(1)!);
        }
        streamUrl??=firstFromMap(data,['mpd_url','m3u8_url','url','stream_url']);
        final responseCookie=(data['cookie']??'').toString().trim();
        if(responseCookie.isNotEmpty)requestHeaders['Cookie']=responseCookie;
        if(streamUrl==null||!isHttp(streamUrl!)) {
          throw Exception('Resolver response mein valid stream URL nahi mila. Fields: '+data.keys.join(', '));
        }
      } catch(e) {
        if(!mounted)return;
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content:Text(channel.name+': '+e.toString()),duration:const Duration(seconds:8)));
        return;
      }
    }
    if(!mounted||streamUrl==null)return;
    await Navigator.of(context).push(MaterialPageRoute(
      builder:(_)=>AnasTvPlayerScreen(channel:channel,url:streamUrl!,headers:requestHeaders,drmLicenseUrl:drmLicenseUrl),
    ));
  }

  // Returns the first balanced top-level JSON object/array and ignores
  // any plain-text footer after it. Brackets inside quoted strings are ignored.
  String _extractJsonValue(String raw) {
    final text=raw.replaceFirst('\uFEFF','').trimLeft();
    final arrayStart=text.indexOf('[');
    final objectStart=text.indexOf('{');
    int start;
    if(arrayStart<0) {
      start=objectStart;
    } else if(objectStart<0) {
      start=arrayStart;
    } else {
      start=arrayStart<objectStart?arrayStart:objectStart;
    }
    if(start<0)return text;

    final opening=text[start];
    final closing=opening=='['?']':'}';
    var depth=0;
    var inString=false;
    var escaped=false;
    for(var i=start;i<text.length;i++) {
      final ch=text[i];
      if(inString) {
        if(escaped) {
          escaped=false;
        } else if(ch==r'\') {
          escaped=true;
        } else if(ch=='"') {
          inString=false;
        }
        continue;
      }
      if(ch=='"') {
        inString=true;
      } else if(ch==opening) {
        depth++;
      } else if(ch==closing) {
        depth--;
        if(depth==0)return text.substring(start,i+1);
      }
    }
    throw const FormatException('Resolver JSON value is incomplete.');
  }

  String? firstFromMap(Map<String,dynamic> data,List<String> keys) {
    for(final key in keys) {
      final value=(data[key]??'').toString().trim();
      if(value.startsWith('https://')||value.startsWith('http://'))return value;
    }
    return null;
  }

            :filtered.isEmpty

                  );
                }),
              ),
      ),
    );
  }

class AnasTvPlayerScreen extends StatefulWidget {
  final TvChannel channel;
  final String url;
  final Map<String,String> headers;
  final String drmLicenseUrl;
  const AnasTvPlayerScreen({super.key,required this.channel,required this.url,required this.headers,this.drmLicenseUrl=''});
  @override State<AnasTvPlayerScreen> createState()=>_AnasTvPlayerScreenState();
}
class _AnasTvPlayerScreenState extends State<AnasTvPlayerScreen> {
  String? _error;
  String _status='Creating native player...';
  bool _loading=true;
  MethodChannel? _channel;
  Timer? _watchdog;

  void _created(int id) {
    final ch=MethodChannel('anas_tv/media3_player_$id');
    _channel=ch;
    if(mounted)setState(()=>_status='Native player created; connecting...');
    ch.setMethodCallHandler((call) async {
      if(!mounted)return;
      if(call.method=='onStateChanged') {
        final state=(call.arguments??'unknown').toString();
        setState(() {
          _status='ExoPlayer: $state';
          _loading=state=='idle'||state=='buffering';
        });
        if(state=='ready'||state=='ended')_watchdog?.cancel();
      } else if(call.method=='onError') {
        _watchdog?.cancel();
        setState(() {
          _loading=false;
          _status='Playback failed';
          _error=(call.arguments??'Native player reported an unspecified error').toString();
        });
      }
    });
    _watchdog?.cancel();
    _watchdog=Timer(const Duration(seconds:20),(){
      if(!mounted||!_loading)return;
      setState(() {
        _loading=false;
        _status='Native player did not report READY or ERROR';
        _error='Playback diagnostic timeout (20s). URL: ${widget.url}\nNative player ne state/error callback nahi bheja. Iska matlab player start command ya platform-view connection fail hua ho sakta hai.';
      });
    });
    ch.invokeMethod('play',{
      'streamUrl':widget.url,
      'headers':widget.headers,
      'drmLicenseUrl':widget.drmLicenseUrl,
    }).catchError((Object e) {
      if(!mounted)return;
      _watchdog?.cancel();
      setState(() {
        _loading=false;
        _status='MethodChannel command failed';
        _error='Native player start failed: $e';
      });
    });
  }

  @override void initState() {
    super.initState();
    // Guard platform-view creation itself. The previous watchdog only started
    // inside _created(), leaving this screen loading forever if the native
    // Android view was never registered or created.
    _watchdog = Timer(const Duration(seconds:15), () {
      if (!mounted || _channel != null) return;
      setState(() {
        _loading = false;
        _status = 'Native platform view was not created';
        _error = 'Flutter did not create the Android native player view within 15 seconds. The Media3 factory may not be registered in MainActivity, or this APK may be outdated. Reinstall the latest successful GitHub Actions artifact.\\nView type: anas_tv/media3_player';
      });
    });
    WidgetsBinding.instance.addPostFrameCallback((_) async {
      await SystemChrome.setEnabledSystemUIMode(SystemUiMode.immersiveSticky);
      if(!mounted)return;
      if(MediaQuery.sizeOf(context).width<700) {
        await SystemChrome.setPreferredOrientations(const [DeviceOrientation.landscapeLeft,DeviceOrientation.landscapeRight]);
      }
    });
  }

  @override void dispose() {
    _watchdog?.cancel();
    _channel?.setMethodCallHandler(null);
    SystemChrome.setEnabledSystemUIMode(SystemUiMode.edgeToEdge);
    SystemChrome.setPreferredOrientations(const [DeviceOrientation.portraitUp]);
    super.dispose();
  }

  @override Widget build(BuildContext context)=>Scaffold(
    backgroundColor:Colors.black,
    body:Stack(fit:StackFit.expand,children:[
      PlatformViewLink(
        key:ValueKey(widget.url),
        viewType:'anas_tv/media3_player',
        surfaceFactory:(context,controller)=>AndroidViewSurface(
          controller:controller as AndroidViewController,
          gestureRecognizers:const <Factory<OneSequenceGestureRecognizer>>{},
          hitTestBehavior:PlatformViewHitTestBehavior.opaque,
        ),
        onCreatePlatformView:(params){
          return PlatformViewsService.initExpensiveAndroidView(
            id:params.id,
            viewType:'anas_tv/media3_player',
            layoutDirection:TextDirection.ltr,
            creationParams:{
              'streamUrl':widget.url,
              'headers':widget.headers,
              'drmLicenseUrl':widget.drmLicenseUrl,
            },
            creationParamsCodec:const StandardMessageCodec(),

        ]),
      )),
      SafeArea(child:Align(alignment:Alignment.topLeft,child:Row(children:[
        IconButton(onPressed:()=>Navigator.of(context).pop(),icon:const Icon(Icons.arrow_back,color:Colors.white)),
        Expanded(child:Text(widget.channel.name,maxLines:1,overflow:TextOverflow.ellipsis,style:const TextStyle(color:Colors.white,fontSize:16,fontWeight:FontWeight.w600))),
      ]))),
    ]),
  );
}

class _ChannelMarquee extends StatefulWidget{
  final String text;
  final TextStyle style;
  const _ChannelMarquee({required this.text,required this.style});
  @override State<_ChannelMarquee> createState()=>_ChannelMarqueeState();
}
class _ChannelMarqueeState extends State<_ChannelMarquee> with SingleTickerProviderStateMixin{
  late final AnimationController _controller;
  bool _running=false;
  double _lastDistance=0;
  static const double _gap=34;
  @override void initState(){
    super.initState();
    _controller=AnimationController(vsync:this);
  }
  @override void didUpdateWidget(covariant _ChannelMarquee oldWidget){
    super.didUpdateWidget(oldWidget);
    if(oldWidget.text!=widget.text){_running=false;_lastDistance=0;_controller.stop();_controller.value=0;}
  }
  @override void dispose(){_controller.dispose();super.dispose();}
  @override Widget build(BuildContext context){
    return LayoutBuilder(builder:(context,constraints){
      final painter=TextPainter(text:TextSpan(text:widget.text,style:widget.style),
        maxLines:1,textDirection:Directionality.of(context))..layout();
      final textWidth=painter.width;
      if(textWidth<=constraints.maxWidth||constraints.maxWidth<=0){
        _running=false;
        _controller.stop();
        return ClipRect(child:Text(widget.text,maxLines:1,softWrap:false,overflow:TextOverflow.ellipsis,
          textAlign:TextAlign.center,style:widget.style));
      }
      final distance=textWidth+_gap;
      if(!_running||(_lastDistance-distance).abs()>1){
        _running=true;_lastDistance=distance;
        _controller.duration=Duration(milliseconds:((distance/38)*1000).round().clamp(1800,18000).toInt());
        WidgetsBinding.instance.addPostFrameCallback((_){
          if(mounted&&_running)_controller.repeat();
        });
      }
      return ClipRect(
        child:AnimatedBuilder(
          animation:_controller,
          builder:(context,child)=>Transform.translate(
            offset:Offset(-distance*_controller.value,0),
            child:child,
          ),
          child:SizedBox(
            width:distance+textWidth,
            child:Row(mainAxisSize:MainAxisSize.min,children:[
              SizedBox(width:textWidth,child:Text(widget.text,maxLines:1,softWrap:false,style:widget.style)),
              const SizedBox(width:_gap),
              SizedBox(width:textWidth,child:Text(widget.text,maxLines:1,softWrap:false,style:widget.style)),
            ]),
          ),
        ),
      );
    });
  }
}

class _ChannelLogo extends StatefulWidget{
  final String primaryUrl,fallbackUrl;
  const _ChannelLogo({required this.primaryUrl,required this.fallbackUrl});
  @override State<_ChannelLogo> createState()=>_ChannelLogoState();
}
class _ChannelLogoState extends State<_ChannelLogo>{
  bool fallback=false;
  @override Widget build(BuildContext context){
    final url=fallback?widget.fallbackUrl:widget.primaryUrl;
    if(url.isEmpty)return const Center(child:Icon(Icons.tv_rounded,size:30,color:Colors.grey));
    return Image.network(url,fit:BoxFit.contain,errorBuilder:(_,__,___){
      if(!fallback&&widget.fallbackUrl.isNotEmpty&&widget.fallbackUrl!=widget.primaryUrl){
        WidgetsBinding.instance.addPostFrameCallback((_){if(mounted)setState(()=>fallback=true);});
      }
      return const Center(child:Icon(Icons.tv_rounded,size:30,color:Colors.grey));
    });
  }
}
