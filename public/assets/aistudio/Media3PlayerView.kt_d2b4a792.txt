package com.anas.tv

import android.content.Context
import android.graphics.Color
import android.view.View
import android.widget.FrameLayout
import android.widget.TextView
import android.os.Handler
import android.os.Looper
import androidx.media3.common.C
import androidx.media3.common.MediaItem
import androidx.media3.common.PlaybackException
import androidx.media3.common.Player
import androidx.media3.datasource.DefaultHttpDataSource
import androidx.media3.exoplayer.ExoPlayer
import androidx.media3.exoplayer.source.DefaultMediaSourceFactory
import androidx.media3.ui.PlayerView
import io.flutter.plugin.common.BinaryMessenger
import io.flutter.plugin.common.MethodChannel
import io.flutter.plugin.platform.PlatformView

class Media3PlayerView(
    context: Context,
    params: Map<String, Any?>,
    messenger: BinaryMessenger,
    viewId: Int
) : PlatformView {
    private val root = FrameLayout(context)
    private val playerView = PlayerView(context)
    private val channel = MethodChannel(messenger, "anas_tv/media3_player_$viewId")
    private val errorView = TextView(context)
    private var player: ExoPlayer? = null

    private val mainHandler = Handler(Looper.getMainLooper())

    init {
        channel.setMethodCallHandler { call, result ->
            when (call.method) {
                "play" -> {
                    @Suppress("UNCHECKED_CAST")
                    val playParams = call.arguments as? Map<String, Any?> ?: emptyMap()
                    val url = (playParams["streamUrl"] as? String).orEmpty().trim()
                    if (url.isBlank()) sendError("Stream URL missing in play parameters.")
                    else startPlayer(context, url, playParams)
                    result.success(null)
                }
                "pause" -> { player?.playWhenReady = false; result.success(null) }
                "resume" -> { player?.playWhenReady = true; result.success(null) }
                else -> result.notImplemented()
            }
        }
        root.setBackgroundColor(Color.BLACK)
        playerView.useController = true
        root.addView(playerView, FrameLayout.LayoutParams(
            FrameLayout.LayoutParams.MATCH_PARENT,
            FrameLayout.LayoutParams.MATCH_PARENT
        ))
        errorView.setTextColor(Color.WHITE)
        errorView.textSize = 14f
        errorView.setPadding(28, 28, 28, 28)
        errorView.setBackgroundColor(0xCC000000.toInt())
        errorView.visibility = View.GONE
        root.addView(errorView, FrameLayout.LayoutParams(
            FrameLayout.LayoutParams.MATCH_PARENT,
            FrameLayout.LayoutParams.WRAP_CONTENT
        ))


    }

    private fun startPlayer(context: Context, streamUrl: String, params: Map<String, Any?>) {
        try {
            @Suppress("UNCHECKED_CAST")
            val headers = (params["headers"] as? Map<String, String>) ?: emptyMap()
            val httpFactory = DefaultHttpDataSource.Factory()
                .setAllowCrossProtocolRedirects(true)
                .setUserAgent(headers["User-Agent"] ?: "AnasTV Android")
                .setDefaultRequestProperties(headers)

            val mediaSourceFactory = DefaultMediaSourceFactory(context)
                .setDataSourceFactory(httpFactory)

            val builder = ExoPlayer.Builder(context)
                .setMediaSourceFactory(mediaSourceFactory)

            val exoPlayer = builder.build()
            player = exoPlayer
            playerView.player = exoPlayer

            val itemBuilder = MediaItem.Builder().setUri(streamUrl)
            val lower = streamUrl.substringBefore("?").lowercase()
            when {
                lower.endsWith(".mpd") -> itemBuilder.setMimeType("application/dash+xml")
                lower.endsWith(".m3u8") -> itemBuilder.setMimeType("application/x-mpegURL")
            }

            // Widevine is enabled only when the resolver supplies an authorized
            // license-server URL. No local keys or key extraction are accepted.
            val licenseUrl = (params["drmLicenseUrl"] as? String).orEmpty()
            if (licenseUrl.startsWith("https://") || licenseUrl.startsWith("http://")) {
                val drmHeaders = (params["drmHeaders"] as? Map<String, String>) ?: emptyMap()
                itemBuilder.setDrmConfiguration(
                    MediaItem.DrmConfiguration.Builder(C.WIDEVINE_UUID)
                        .setLicenseUri(licenseUrl)
                        .setLicenseRequestHeaders(drmHeaders)
                        .build()
                )
            }

            exoPlayer.addListener(object : Player.Listener {
                override fun onPlayerError(error: PlaybackException) {
                    val causes = generateSequence(error as Throwable?) { it.cause }
                        .take(8)
                        .map { "${it.javaClass.simpleName}: ${it.message ?: "(no message)"}" }
                        .joinToString("\\nCaused by: ")
                    val message = "Media3 playback failed\\nCode: ${error.errorCodeName} (${error.errorCode})\\n${error.message ?: "No player message"}\\nCause chain: ${causes}"
                    sendError(message)
                }
                override fun onPlaybackStateChanged(playbackState: Int) {
                    val state = when (playbackState) {
                        Player.STATE_BUFFERING -> "buffering"
                        Player.STATE_READY -> "ready"
                        Player.STATE_ENDED -> "ended"
                        else -> "idle"
                    }
                    sendState(state)
                    if (playbackState == Player.STATE_READY) errorView.visibility = View.GONE
                }
                override fun onRenderedFirstFrame() {
                    sendState("ready")
                    errorView.visibility = View.GONE
                }
            })
            exoPlayer.setMediaItem(itemBuilder.build())
            exoPlayer.prepare()
            exoPlayer.playWhenReady = true
        } catch (error: Exception) {
            showError("Player setup error: ${error.message ?: error.javaClass.simpleName}")
        }
    }

    private fun sendError(message: String) {
        mainHandler.post {
            // Keep native overlay hidden; Flutter owns the playback error UI.
            errorView.visibility = View.GONE
            channel.invokeMethod("onError", message)
        }
    }

    private fun sendState(state: String) {
        mainHandler.post { channel.invokeMethod("onStateChanged", state) }
    }

    private fun showError(message: String) = sendError(message)

    override fun getView(): View = root

    override fun dispose() {
        channel.setMethodCallHandler(null)
        playerView.player = null
        player?.release()
        player = null
    }
}
