name: Build Android APK

on:
  workflow_dispatch:
  push:
    branches: [main]
    paths:
      - 'lib/**'
      - 'android_native/**'
      - 'pubspec.yaml'
      - '.github/workflows/build-apk.yml'

permissions:
  contents: read

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout source
        uses: actions/checkout@v4

      - name: Set up Java
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - name: Set up Flutter
        uses: subosito/flutter-action@v2
        with:
          channel: stable
          cache: true

      - name: Generate Android project files if missing
        run: |
          if [ ! -d android ]; then
            flutter create --platforms=android --org com.anas.tv .
          fi

      - name: Install native Media3 player
        run: |
          mkdir -p android/app/src/main/kotlin/com/anas/tv
          cp android_native/MainActivity.kt android/app/src/main/kotlin/com/anas/tv/MainActivity.kt
          cp android_native/Media3PlayerView.kt android/app/src/main/kotlin/com/anas/tv/Media3PlayerView.kt
          python3 - <<'PY'
          from pathlib import Path
          candidates = [Path("android/app/build.gradle.kts"), Path("android/app/build.gradle")]
          gradle = next((p for p in candidates if p.exists()), None)
          if gradle is None:
              raise SystemExit("Could not find Android app Gradle file")
          s = gradle.read_text()
          # The pubspec project name is "anas_tv". Flutter otherwise generates
          # namespace/applicationId "com.anas.tv.anas_tv", while our native
          # MainActivity is deliberately packaged as "com.anas.tv". Align the
          # Android app identity so the manifest launches the Activity that
          # registers the Media3 platform-view factory.
          import re
          s = re.sub(r'(?m)^\s*namespace\s*=\s*["\'][^"\']+["\']',
                     '    namespace = "com.anas.tv"', s)
          s = re.sub(r'(?m)^\s*applicationId\s*=\s*["\'][^"\']+["\']',
                     '        applicationId = "com.anas.tv"', s)
          deps = [
              '    implementation("androidx.media3:media3-exoplayer:1.6.1")',
              '    implementation("androidx.media3:media3-exoplayer-dash:1.6.1")',
              '    implementation("androidx.media3:media3-exoplayer-hls:1.6.1")',
              '    implementation("androidx.media3:media3-ui:1.6.1")',
          ]
          if "androidx.media3:media3-exoplayer:" not in s:
              block = "dependencies {\n" + "\n".join(deps) + "\n"
              if "dependencies {" in s:
                  s = s.replace("dependencies {", block, 1)
              else:
                  s += "\n" + block + "}\n"
          gradle.write_text(s)
          PY

      - name: Ensure Android network permissions
        run: |
          python3 - <<'PY'
          from pathlib import Path
          p = Path("android/app/src/main/AndroidManifest.xml")
          s = p.read_text()
          permissions = [
              '    <uses-permission android:name="android.permission.INTERNET" />',
              '    <uses-permission android:name="android.permission.ACCESS_NETWORK_STATE" />',
          ]
          for permission in permissions:
              if permission not in s:
                  s = s.replace('<application', permission + '\n    <application', 1)
          p.write_text(s)
          PY

      - name: Install dependencies
        run: flutter pub get

      - name: Build release APK
        run: flutter build apk --release

      - name: Upload APK
        uses: actions/upload-artifact@v4
        with:
          name: anas-tv-release-apk
          path: build/app/outputs/flutter-apk/app-release.apk
          if-no-files-found: error
          retention-days: 14
